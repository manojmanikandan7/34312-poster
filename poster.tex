    \begin{exampleblock}{Introduction}
    Gradient Descent, originally discussed by \cite{cauchy1847methode}, is the simplest strategy to minimize a differentiable function
    $f$ on $\mathbb{R}_n$. It is well-known that the convergence rate for this algorithm is
    $O(1/t)$. Here, we discuss the lower bounds on the convergence rates for $\beta$-smooth convex functions in
    $O(1/t^2)$, and emphasize how Nesterov's Accelerated Gradient Descent achieves this bound. The main source of reference for this poster is from Chapter 3 of
    \cite{bubeck2015convex}.
    \end{exampleblock}

    \begin{exampleblock}{Notation}
      $\delta f(x)$ is the set of \textbf{subgradients} of the point $x \in \mathbb{R}^n$. Then the subgradient $g \in \delta f(x)$ satisfies $f(x) - f(y) \leq g^\top (x-y)$. \\
      $e_1, \dots, e_n$ is the canonical basis of $\mathbb{R}^n$. \\
      $x^* \in \mathbb{R}^n$ is a point such that it is the minima in any optimisation problem, i.e. $f(x^*) = \mathop{\min}\limits_{x \in \mathbb{R}^n} f(x)$.
      We always assume $x^*$ exists. \\
      $\norm{\cdot}$ is always the Euclidean norm $\norm{x} = \sqrt{x_1^2 + x_2^2 + \dots + x_n^2}$ for $x \in \mathbb{R}^n$
    \end{exampleblock}
   \begin{exampleblock}{Fastest theoretical speed for Gradient Descent}
   Theorem (\ref{thm:3.14}) states that, no matter the gradient method provided, there
   always exists a function $f$ such that when the gradient method is applied on
   minimizing such $f$, the convergence rate is lower bounded as $O(1/t^2)$.
    \begin{theorem}[{\bf 1}] \label{thm:3.14}
      Let $t \leq \frac{n-1}{2}$ and $\beta > 0$. There exists a $\beta$-smooth convex function $f$ such that for any black-box procedure satisfying
      $x_{t+1} \in \text{Span}(g_1, \dots, g_t)$ with $g_s \in \delta f(x_s)$:
      \begin{equation*}
        \mathop{\min}\limits_{1 \leq s \leq t} f(x_s) - f(x^*) \geq \frac{3\beta}{32} \frac{\norm{x_1 - x^*}^2}{(t + 1)^2}
      \end{equation*}
    \end{theorem}
    \begin{center}
      \textcolor{red}{We observe that Theorem~\ref{thm:3.19} attains this complexity bound of $O(1/t^2)$}
    \end{center}
    \end{exampleblock}

    \begin{exampleblock}{Crafting a $\beta$-smooth function $f$}
      To construct a suitable function $f$ that satisfies (subsequently, proves) Theorem (\ref{thm:3.14}), \cite{bubeck2015convex} constructs a symmetric and tridiagonal matrix
      \vspace{-2mm}
      \begin{equation*}
        (A_k)_{i,j} = \begin{cases}
          2, \ &i=j \ , \ i \leq k \\
          -1, \ &j \in \{i - 1, i + 1\}, \ i \leq k, \ j \neq k + 1 \\
          0, &\text{ otherwise}
        \end{cases}
      \end{equation*}
      \vspace{-2mm}
      and defines $f$ as:
      \vspace{-2mm}
      \begin{equation*}
        f_s(x) = \frac{\beta}{8}x^\top A_s x - \frac{\beta}{4}x^\top e_1
      \end{equation*}
      \vspace{-2mm}
      For $s \le t$, we have $x_s(i) = 0$ for $i = s,\ldots, n; n=2t+1$, which implies $x_s^\top A_{n} x_s \ge x_s^\top A_s x_s$. This implies,
      \vspace{-2mm}
      \begin{equation}\label{eq:1}
        f(x_s) - f^* = f(x_s) - f^*_{n} \geq f^*_s - f^*_{2t + 1} \geq f^*_t - f^*_{2t + 1}
      \end{equation}
      \vspace{-1mm}
      $x^*$ is the unique solution in the span of $e_1, \ldots , e_k$ of
      $A_k x = e_1$. It is easy to verify that it is defined by $x^*_k(i) = 1 - \frac{i}{k+1}$ for $i = 1, \ldots , k$. Thus,
      \vspace{-2mm}
      \begin{equation}\label{eq:2}
        f^*_k = \frac{\beta}{8}(x^*_k)^\top A_k x^*_k - \frac{\beta}{4}(x^*_k)^\top e_1 = - \frac{\beta}{8}(x^*_k)^\top e_1 = -\frac{\beta}{8} \left(1 - \frac{1}{k+1}\right)
      \end{equation}
      \vspace{-1mm}
      From definition of $x^*_k$ for $k=n$, and $x_1=0$, we can derive lower bounds for $n \ge \frac{3}{2}||x_1 - x^*_n||_2^2$.
      This bound, along with substitution of Equation \ref{eq:2} in Equation \ref{eq:1} resolves to Theorem (\ref{thm:3.14}).

    \end{exampleblock}

\begin{exampleblock}{Descent Lemma}
The following lemma establishes the per-step bound for gradient descent on smooth functions. The original
paper states the constrained case but for Nesterov's accelerated gradient descent in the smooth
function case, we only need the unconstrained version.
\begin{lemma}[\bf 2]\label{lemma:3.6}
Let $f:\mathbb{R}^n \rightarrow \mathbb{R}$ be a convex and $\beta$-smooth function. Let $x, y \in \mathbb{R}^n$ and $x^+ = x - \frac{1}{\beta}\nabla f(x)$.
Then the following holds true:
\begin{equation*}
  f(x^+) - f(y) \leq \nabla f(x)^\top (x - y) - \frac{1}{2\beta}\norm{\nabla f(x)}^2
\end{equation*}
\end{lemma}
\end{exampleblock}




\begin{exampleblock}{Nesterov's accelerated gradient descent}

    \begin{figure}
      \begin{center}
        \begin{tikzpicture}[scale=2.5]
          \node [tokens=1] (noeud1) at (0.5,1) [label=below right:{$x_s$}] {};
          \node [tokens=1] (noeud2) at (1.5,-1) [label=below right:{$y_{s}$}] {};
          \node [tokens=1] (noeud3) at (2.5,2) [label=below right:{$y_{s+1}$}] {};
          \node [tokens=1] (noeud4) at (2.8,3) [label=above left:{$x_{s+1}$}] {};
          \draw[->, thick] (noeud1) -- (noeud3) node[midway, left, xshift=-0.3cm] {$-\frac{1}{\beta}\nabla f(x_s)$};
          \draw[thick, dashed] (noeud2) -- (noeud4) {};
          \node [tokens=1] (noeud5) at (4.5,3.3) [label=below right:{$y_{s+2}$}] {};
          \node [tokens=1] (noeud6) at (5.3,3.8) [label=right:{$x_{s+2}$}] {};
          \draw[->, thick] (noeud4) -- (noeud5) {};
          \draw[thick, dashed] (noeud3) -- (noeud6) {};
        \end{tikzpicture}
        \end{center}
      \caption{Illustration of Nesterov's accelerated gradient descent.}
      \label{fig:nesterovacc}
    \end{figure}
    First we introduce the following sequences:
    \begin{equation*}
      \lambda_0 = 0, \ \lambda_t = \frac{1 + \sqrt{1 + 4\lambda_{t-1}^2}}{2}, \ \text{and} \ \gamma_t = \frac{1 - \lambda_t}{\lambda_{t + 1}}
    \end{equation*}
    Nesterov's accelerated gradient descent is described as follows. Start at an arbitrary initial point $x_1=y_1$ and then iterate
    the following equations for $t\geq1$
    \begin{align}
    y_{t+1} &= x_t - \frac{1}{\beta}\nabla f(x_t) \label{nesterov:y}\\
    x_{t+1} &= (1 - \gamma_t)y_{t+1} + \gamma_t y_t \label{nesterov:x}
    \end{align}
    Nesterov's accelerated gradient descent uses the ideas from classical momentum gradient descent \parencite{sutskever}.
    $x_{t+1}$ can be rewritten as $x_{t+1} = y_{t+1} - \gamma_t(y_{t+1} - y_{t})$ which reads as $x_{t+1}$ overshoots $y_{t+1}$ along the vector
    $(y_{t+1} - y_{t})$ with some ``momentum'' given by $\gamma_t$.
    \begin{theorem}[{\bf 3}]\label{thm:3.19}
    Let $f$ be a convex and $\beta$-smooth function, then Nesterov's accelerated gradient descent satisfies
    \newline \\
    \begin{equation*}
      f(y_t) - f(x^*) \leq \frac{2\beta\norm{x_1 - x^*}^2}{t^2}
    \end{equation*}
    \end{theorem}

    This gives us the $O(1/t^2)$ complexity. \\
    \textit{Achieving this accelerated rate relies on carefully choosing the parameters
    $\lambda_t$ and $\gamma_t$ so that the convergence inequalities telescope across iterations.}
    \end{exampleblock}

    % ============================================================
    % CORRECTED PANEL — Key insights from Proof of Theorem 3
    % ============================================================
    \begin{exampleblock}{Key insights from Proof of Theorem $3$}
    \begin{itemize}
      %
      % --- SETUP: define d_t and g_t ---
      %
      \item We can reverse engineer both $\lambda_t$ and $\gamma_t$ to
      show they are not arbitrary.
      Define $d_t := x_t - y_t$ (the momentum displacement) and
      $g_t := -\tfrac{1}{\beta}\nabla f(x_t)$ (the gradient step),
      so that $y_{t+1} = x_t + g_t$.
      Applying Lemma~\ref{lemma:3.6} with $x = x_t$ and
      $\nabla f(x_t) = -\beta g_t$ gives two inequalities:
      \begin{align*}
        f(y_{t+1}) - f(y_t)
          &\leq -\frac{\beta}{2}\bigl(\norm{g_t}^2
            + 2\,g_t \cdot d_t\bigr)
          \tag{$\star$} \label{ineq:1} \\[4pt]
        f(y_{t+1}) - f(x^*)
          &\leq -\frac{\beta}{2}\bigl(\norm{g_t}^2
            + 2\,g_t \cdot (x_t - x^*)\bigr)
          \tag{$\star\!\star$} \label{ineq:2}
      \end{align*}
      %
      % --- COMBINE via identity ---
      %
      \item The big idea is to use the identity
      $\norm{a}^2 + 2\,a \cdot b = \norm{a+b}^2 - \norm{b}^2$
      on the combination of both inequalities to obtain a telescopic
      sum~\parencite{BubeckUpdatePost}.
      Let $\delta_t = f(y_t) - f(x^*)$.
      Multiplying~(\ref{ineq:1}) by $(\lambda_t - 1)$ and
      adding~(\ref{ineq:2}), and using
      $(\lambda_t{-}1)\,d_t + x_t = y_t + \lambda_t d_t$, we obtain:
      \begin{align*}
        \lambda_t\,\delta_{t+1} - (\lambda_t {-} 1)\,\delta_t
          &\leq -\frac{\beta}{2}\bigl(
            \lambda_t\norm{g_t}^2
            + 2\,g_t \cdot (y_t + \lambda_t d_t - x^*)\bigr)
      \end{align*}
      Multiplying both sides by $\lambda_t$ and applying the identity
      with $a = \lambda_t g_t$, \ $b = y_t + \lambda_t d_t - x^*$:
      \begin{align*}
        \lambda_t^2\,\delta_{t+1} - (\lambda_t^2 {-} \lambda_t)\,\delta_t
          &\leq -\frac{\beta}{2}\bigl(
            \norm{y_t + \lambda_t d_t + \lambda_t g_t - x^*}^2 \\[-2pt]
          &\phantom{{}\leq -\frac{\beta}{2}\bigl(}
            -\;\norm{y_t + \lambda_t d_t - x^*}^2\bigr)
          \tag{$\dagger$} \label{eq:dagger}
      \end{align*}
      %
      % --- TELESCOPING: derive gamma_t ---
      %
      \item To obtain the telescopic sum, we need
      \begin{equation*}
        y_t + \lambda_t(d_t + g_t)
          \;=\; y_{t+1} + \lambda_{t+1}\,d_{t+1}
      \end{equation*}
      which comes from observing~(\ref{eq:dagger}). Since $y_{t+1} = y_t + d_t + g_t$ and
      $d_{t+1} = x_{t+1} - y_{t+1} = -\gamma_t(d_t + g_t)$:
      \begin{align*}
        \lambda_t(d_t + g_t)
          &= (d_t + g_t) + \lambda_{t+1}\bigl[-\gamma_t(d_t + g_t)\bigr] \\
        \lambda_t(d_t + g_t)
          &= (d_t + g_t)(1 - \lambda_{t+1}\,\gamma_t)
      \end{align*}
      and hence
      $\;\gamma_t = \dfrac{1 - \lambda_t}{\lambda_{t+1}}$.
      %
      % --- SELECTING lambda_t ---
      %
      \item  By denoting
      $u_t := \tfrac{\beta}{2}\norm{y_t + \lambda_t d_t - x^*}^2$, with~(\ref{eq:dagger}) now telescoping as
      $\lambda_t^2\,\delta_{t+1} - (\lambda_t^2 {-} \lambda_t)\,\delta_t
      \leq u_t - u_{t+1}$, we require
      $\lambda_t^2 - \lambda_t = \lambda_{t-1}^2$ for the left side
      to collapse. Summing from $t=1$ to $T{-}1$ and using
      $u_1 = \tfrac{\beta}{2}\norm{x_1 - x^*}^2$
      (since $d_1 = 0$) and $\lambda_{T-1} \geq T/2$
      completes the proof.
    \end{itemize}
    \end{exampleblock}
